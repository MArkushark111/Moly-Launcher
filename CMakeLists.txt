cmake_minimum_required(VERSION 3.24)
project(MolyLauncher VERSION 0.8.5 LANGUAGES CXX)
option(MOLY_LOCAL_EDITION "Build the personal, game-download-free local edition" OFF)
option(MOLY_PUBLIC_BUILD "Build without private keys or bundled Minecraft textures" OFF)
option(MOLY_BUILD_RUST_ENGINE "Compile the Rust engine during the launcher build" ON)
set(CMAKE_INSTALL_BINDIR ".")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core CorePrivate Gui Quick QuickControls2 Network Concurrent Sql Test)
qt_standard_project_setup(REQUIRES 6.11)
add_library(moly_engine STATIC
    src/localedition.cpp src/localedition.h src/core.cpp src/core.h src/rustvisuals.cpp src/rustvisuals.h src/network.cpp src/network.h
    src/minecraft.cpp src/minecraft.h src/accounts.cpp src/accounts.h
    src/content.cpp src/content.h src/packexport.cpp src/packexport.h)
target_include_directories(moly_engine PUBLIC src)
if(MOLY_LOCAL_EDITION)
    target_compile_definitions(moly_engine PUBLIC MOLY_LOCAL_EDITION=1)
endif()
target_link_libraries(moly_engine PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent PRIVATE Qt6::CorePrivate)
if(WIN32)
    target_link_libraries(moly_engine PRIVATE advapi32)
endif()
add_library(moly_desktop STATIC src/releaseupdates.cpp src/releaseupdates.h src/discordpresence.cpp src/discordpresence.h src/instanceimport.cpp src/instanceimport.h src/fileassociation.cpp src/fileassociation.h src/artwork.cpp src/artwork.h src/playtargets.cpp src/playtargets.h src/skinstore.cpp src/skinstore.h src/skinmodel.cpp src/skinmodel.h)
set(MOLY_MICROSOFT_CLIENT_ID "8fa8a03a-5fd4-4b62-a547-7eb20028cf4e" CACHE STRING "Moly's registered Microsoft public application ID")
if(MOLY_MICROSOFT_CLIENT_ID STREQUAL "")
    set(MOLY_MICROSOFT_CLIENT_ID "8fa8a03a-5fd4-4b62-a547-7eb20028cf4e" CACHE STRING "Moly Microsoft application ID" FORCE)
endif()
set(MOLY_DISCORD_APPLICATION_ID "1564996413849346068" CACHE STRING "Moly's registered Discord application ID")
set(MOLY_CURSEFORGE_API_KEY "" CACHE STRING "Moly's bundled CurseForge API key")
# The CurseForge key is a private secret: it is kept out of the repository and loaded from this
# untracked file when present, so release builds bundle it while clones and forks stay keyless.
if(NOT MOLY_PUBLIC_BUILD AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/local-secrets.cmake")
    include("${CMAKE_CURRENT_SOURCE_DIR}/local-secrets.cmake")
endif()
if(MOLY_LOCAL_EDITION OR MOLY_PUBLIC_BUILD)
    set(MOLY_CURSEFORGE_API_KEY "")
endif()
target_compile_definitions(moly_desktop PUBLIC MOLY_MICROSOFT_CLIENT_ID="${MOLY_MICROSOFT_CLIENT_ID}" MOLY_DISCORD_APPLICATION_ID="${MOLY_DISCORD_APPLICATION_ID}" MOLY_CURSEFORGE_API_KEY="${MOLY_CURSEFORGE_API_KEY}")
target_include_directories(moly_desktop PUBLIC src)
target_link_libraries(moly_desktop PUBLIC moly_engine Qt6::Gui Qt6::Quick Qt6::Network Qt6::Concurrent Qt6::Sql PRIVATE Qt6::CorePrivate)
if(MOLY_PUBLIC_BUILD)
    target_compile_definitions(moly_engine PUBLIC MOLY_PUBLIC_BUILD=1)
else()
file(GLOB default_skin_textures CONFIGURE_DEPENDS "assets/default-skins/wide/*.png" "assets/default-skins/slim/*.png")
qt_add_resources(moly_desktop "default_skins" PREFIX "/moly/skins" BASE "assets/default-skins" FILES ${default_skin_textures})
endif()
qt_add_executable(MolyLauncher WIN32 src/startup.h src/main.cpp src/launcher.cpp src/launcher.h)
if(MOLY_LOCAL_EDITION)
    set_target_properties(MolyLauncher PROPERTIES OUTPUT_NAME "MolyLocalLauncher")
endif()
if(WIN32)
    enable_language(RC)
    target_sources(MolyLauncher PRIVATE assets/moly.rc)
    target_include_directories(MolyLauncher PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(MolyLauncher URI Moly VERSION 1.0 QML_FILES
    qml/LauncherUpdates.qml qml/LoginStatus.qml qml/Intro.qml qml/AccountChooser.qml qml/ServerDialog.qml qml/ContentBrowserWindow.qml qml/ContentVersionsDialog.qml qml/Startup.qml qml/ImportInstancesDialog.qml qml/ReorderHandle.qml qml/MSpinBox.qml qml/ArcSpinner.qml qml/MSlider.qml qml/MComboBox.qml qml/ProviderIcon.qml qml/MMenu.qml qml/MMenuItem.qml qml/MCheckBox.qml qml/TaskActivity.qml qml/RustVisualsPanel.qml qml/InstanceSettings.qml qml/MProgressBar.qml qml/LiquidDialog.qml qml/InstanceAvatar.qml qml/Theme.qml qml/MSwitch.qml qml/Icon.qml qml/NavItem.qml qml/TabStrip.qml qml/InstallDialog.qml
    qml/Main.qml qml/MButton.qml qml/MField.qml qml/Panel.qml qml/Badge.qml
    qml/LocalLibraryPage.qml qml/LibraryPage.qml qml/InstancePage.qml qml/DiscoverPage.qml
    qml/AccountsPage.qml qml/SettingsPage.qml qml/LogView.qml qml/LogWindow.qml
    qml/MediaImage.qml qml/SkinsPage.qml qml/PlayTargets.qml qml/CreateInstanceDialog.qml qml/DownloadManager.qml
    RESOURCES assets/dawn.png assets/fastclient.svg assets/norisk.png assets/badlion.png assets/labymod.png assets/feather.png assets/gdlauncher.ico assets/prism.svg assets/multimc.svg assets/atlauncher.png assets/lunar.ico assets/minecraft-folder.svg assets/modrinth.svg assets/curseforge.svg assets/moly.svg assets/moly-monochrome.svg assets/Inter.ttf assets/Inter-OFL.txt)
target_link_libraries(MolyLauncher PRIVATE moly_desktop Qt6::QuickControls2)

if(MOLY_BUILD_RUST_ENGINE AND MOLY_CARGO AND NOT MOLY_LOCAL_EDITION)
    add_custom_target(moly_rust_engine
        COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File "${CMAKE_CURRENT_SOURCE_DIR}/scripts/build-rust-engine.ps1" -BuildDirectory "${CMAKE_BINARY_DIR}"
        WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}" VERBATIM)
    add_dependencies(MolyLauncher moly_rust_engine)
endif()
install(FILES "${CMAKE_BINARY_DIR}/engines/moly-rust.exe" "${CMAKE_BINARY_DIR}/engines/moly-rust-prepare.exe" DESTINATION engines OPTIONAL)
install(DIRECTORY "${CMAKE_BINARY_DIR}/engines/horizon-support/" DESTINATION engines/horizon-support OPTIONAL)

install(TARGETS MolyLauncher RUNTIME DESTINATION .)
if(NOT MOLY_LOCAL_EDITION AND NOT MOLY_PUBLIC_BUILD)
    qt_generate_deploy_qml_app_script(TARGET MolyLauncher OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
    install(SCRIPT ${deploy_script})
endif()

    install(FILES "${CMAKE_BINARY_DIR}/imageformats/qwebp.dll" DESTINATION plugins/imageformats OPTIONAL)

# Shared Qt plugins are deployed with windeployqt --qmldir qml for the local package.
# Avoid scanning the multi-gigabyte private library and vendored engine for each test.
if(MOLY_LOCAL_EDITION OR MOLY_PUBLIC_BUILD)
    get_property(moly_local_targets DIRECTORY PROPERTY BUILDSYSTEM_TARGETS)
    foreach(local_target IN LISTS moly_local_targets)
        set_property(TARGET ${local_target} PROPERTY QT_QML_MODULE_NO_IMPORT_SCAN TRUE)
    endforeach()
endif()
