cmake_minimum_required(VERSION 3.24)
project(MolyLauncher VERSION 0.7.0 LANGUAGES CXX)
set(CMAKE_INSTALL_BINDIR ".")
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.11 REQUIRED COMPONENTS Core CorePrivate Gui Quick QuickControls2 Network Concurrent Sql Test)
qt_standard_project_setup(REQUIRES 6.11)
add_library(moly_engine STATIC
    src/core.cpp src/core.h src/network.cpp src/network.h
    src/minecraft.cpp src/minecraft.h src/accounts.cpp src/accounts.h
    src/content.cpp src/content.h src/packexport.cpp src/packexport.h)
target_include_directories(moly_engine PUBLIC src)
target_link_libraries(moly_engine PUBLIC Qt6::Core Qt6::Network Qt6::Concurrent PRIVATE Qt6::CorePrivate)
if(WIN32)
    target_link_libraries(moly_engine PRIVATE advapi32)
endif()
add_library(moly_desktop STATIC src/discordpresence.cpp src/discordpresence.h src/instanceimport.cpp src/instanceimport.h src/fileassociation.cpp src/fileassociation.h src/artwork.cpp src/artwork.h src/playtargets.cpp src/playtargets.h src/skinstore.cpp src/skinstore.h src/skinmodel.cpp src/skinmodel.h)
set(MOLY_MICROSOFT_CLIENT_ID "" CACHE STRING "Moly's registered Microsoft public application ID")
set(MOLY_DISCORD_APPLICATION_ID "1564996413849346068" CACHE STRING "Moly's registered Discord application ID")
target_compile_definitions(moly_desktop PUBLIC MOLY_MICROSOFT_CLIENT_ID="${MOLY_MICROSOFT_CLIENT_ID}" MOLY_DISCORD_APPLICATION_ID="${MOLY_DISCORD_APPLICATION_ID}")
target_include_directories(moly_desktop PUBLIC src)
target_link_libraries(moly_desktop PUBLIC moly_engine Qt6::Gui Qt6::Quick Qt6::Network Qt6::Concurrent Qt6::Sql PRIVATE Qt6::CorePrivate)
file(GLOB default_skin_textures CONFIGURE_DEPENDS "assets/default-skins/wide/*.png" "assets/default-skins/slim/*.png")
qt_add_resources(moly_desktop "default_skins" PREFIX "/moly/skins" BASE "assets/default-skins" FILES ${default_skin_textures})
qt_add_executable(MolyLauncher WIN32 src/startup.h src/main.cpp src/launcher.cpp src/launcher.h)
if(WIN32)
    enable_language(RC)
    target_sources(MolyLauncher PRIVATE assets/moly.rc)
    target_include_directories(MolyLauncher PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/assets")
endif()
set_source_files_properties(qml/Theme.qml PROPERTIES QT_QML_SINGLETON_TYPE TRUE)
qt_add_qml_module(MolyLauncher URI Moly VERSION 1.0 QML_FILES
    qml/AccountChooser.qml qml/ServerDialog.qml qml/ContentBrowserWindow.qml qml/ContentVersionsDialog.qml qml/Startup.qml qml/ImportInstancesDialog.qml qml/ReorderHandle.qml qml/MSpinBox.qml qml/ArcSpinner.qml qml/MSlider.qml qml/MComboBox.qml qml/ProviderIcon.qml qml/MMenu.qml qml/MMenuItem.qml qml/MCheckBox.qml qml/TaskActivity.qml qml/InstanceSettings.qml qml/MProgressBar.qml qml/LiquidDialog.qml qml/InstanceAvatar.qml qml/Theme.qml qml/MSwitch.qml qml/Icon.qml qml/NavItem.qml qml/TabStrip.qml qml/InstallDialog.qml
    qml/Main.qml qml/MButton.qml qml/MField.qml qml/Panel.qml qml/Badge.qml
    qml/LibraryPage.qml qml/InstancePage.qml qml/DiscoverPage.qml
    qml/AccountsPage.qml qml/SettingsPage.qml qml/LogView.qml qml/LogWindow.qml
    qml/MediaImage.qml qml/SkinsPage.qml qml/PlayTargets.qml qml/CreateInstanceDialog.qml qml/DownloadManager.qml
    RESOURCES assets/dawn.png assets/fastclient.svg assets/norisk.png assets/badlion.png assets/labymod.png assets/feather.png assets/gdlauncher.ico assets/prism.svg assets/multimc.svg assets/atlauncher.png assets/lunar.ico assets/minecraft-folder.svg assets/modrinth.svg assets/curseforge.svg assets/moly.svg assets/moly-monochrome.svg assets/Inter.ttf assets/Inter-OFL.txt)
target_link_libraries(MolyLauncher PRIVATE moly_desktop Qt6::QuickControls2)
enable_testing()
qt_add_executable(moly_tests tests/core_tests.cpp)
target_link_libraries(moly_tests PRIVATE moly_engine Qt6::Test Qt6::CorePrivate)
add_test(NAME engine COMMAND moly_tests)
qt_add_executable(launcher_tests tests/launcher_tests.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(launcher_tests PRIVATE moly_desktop Qt6::Test)
add_test(NAME launcher_behaviors COMMAND launcher_tests)
qt_add_executable(install_integration tests/install_integration.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(install_integration PRIVATE moly_desktop)
qt_add_executable(creation_integration tests/creation_integration.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(creation_integration PRIVATE moly_desktop)
qt_add_executable(pagination_integration tests/pagination_integration.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(pagination_integration PRIVATE moly_desktop)
qt_add_executable(packexport_tests tests/packexport_tests.cpp)
target_link_libraries(packexport_tests PRIVATE moly_engine Qt6::Test Qt6::CorePrivate)
add_test(NAME portable_packs COMMAND packexport_tests)
qt_add_executable(skin_tests tests/skin_tests.cpp)
target_link_libraries(skin_tests PRIVATE moly_desktop Qt6::Test)
add_test(NAME skins COMMAND skin_tests)
qt_add_executable(playtargets_tests tests/playtargets_tests.cpp)
target_link_libraries(playtargets_tests PRIVATE moly_desktop Qt6::Test)
add_test(NAME play_targets COMMAND playtargets_tests)
qt_add_executable(quickplay_tests tests/quickplay_tests.cpp)
target_link_libraries(quickplay_tests PRIVATE moly_engine Qt6::Test)
add_test(NAME quick_play COMMAND quickplay_tests)
qt_add_executable(export_integration tests/export_integration.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(export_integration PRIVATE moly_desktop)
qt_add_executable(moly_smoke tests/smoke.cpp)
qt_add_executable(pack_benchmark tests/pack_benchmark.cpp)
target_link_libraries(pack_benchmark PRIVATE moly_engine)
target_link_libraries(moly_smoke PRIVATE moly_engine)
install(TARGETS MolyLauncher RUNTIME DESTINATION .)
qt_generate_deploy_qml_app_script(TARGET MolyLauncher OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
install(SCRIPT ${deploy_script})
if(WIN32)
    install(FILES "${CMAKE_BINARY_DIR}/imageformats/qwebp.dll" DESTINATION plugins/imageformats OPTIONAL)
endif()
set(CPACK_PACKAGE_NAME "Moly Launcher")
set(CPACK_PACKAGE_VENDOR "Moly")
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_INSTALL_DIRECTORY "Moly Launcher")
set(CPACK_PACKAGE_EXECUTABLES "MolyLauncher;Moly Launcher")
set(CPACK_GENERATOR "ZIP")
include(CPack)

qt_add_executable(confirm_integration tests/confirm_integration.cpp src/launcher.cpp src/launcher.h)
target_link_libraries(confirm_integration PRIVATE moly_desktop)

qt_add_executable(instanceimport_tests tests/instanceimport_tests.cpp)
target_link_libraries(instanceimport_tests PRIVATE moly_desktop Qt6::Test Qt6::Sql)
add_test(NAME local_instance_import COMMAND instanceimport_tests)

qt_add_executable(artwork_tests tests/artwork_tests.cpp)
target_link_libraries(artwork_tests PRIVATE moly_desktop Qt6::Test)
target_compile_definitions(artwork_tests PRIVATE MOLY_QML_DIR="${CMAKE_CURRENT_SOURCE_DIR}/qml")
add_test(NAME artwork_rendering COMMAND artwork_tests)
qt_add_executable(startup_tests tests/startup_tests.cpp src/startup.h)
target_link_libraries(startup_tests PRIVATE Qt6::Core Qt6::Gui Qt6::Test)
target_include_directories(startup_tests PRIVATE src)
add_test(NAME startup_stages COMMAND startup_tests)

qt_add_executable(discord_tests tests/discord_tests.cpp)
target_link_libraries(discord_tests PRIVATE moly_desktop Qt6::Test)
add_test(NAME discord_presence COMMAND discord_tests)
